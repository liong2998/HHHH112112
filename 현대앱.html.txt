    let currentBrand = '';

    function showList(brand) {
        currentBrand = brand;
        document.getElementById('screen-main').classList.add('hidden');
        document.getElementById('screen-list').classList.remove('hidden');
        document.getElementById('list-title').innerText = brand.toUpperCase() + " 라인업";
        
        const listDiv = document.getElementById('car-list');
        listDiv.innerHTML = '';
        config[brand].cars.forEach(car => {
            const btn = document.createElement('div');
            btn.className = 'car-item';
            btn.innerText = car;
            btn.onclick = () => showDetail(car);
            listDiv.appendChild(btn);
        });
    }

    function showDetail(carName) {
        document.getElementById('screen-list').classList.add('hidden');
        document.getElementById('screen-detail').classList.remove('hidden');
        document.getElementById('detail-title').innerText = carName;

        const info = carData[carName] || { spec: {"안내": "상세 데이터 업데이트 중"}, price: "상담 시 안내", colors: "홈페이지 참고", pdf: "#" };
        
        const specTable = document.getElementById('spec-data');
        specTable.innerHTML = '';
        for (let key in info.spec) {
            specTable.innerHTML += `<tr><td style="color:#888; width:40%">${key}</td><td>${info.spec[key]}</td></tr>`;
        }

        document.getElementById('price-summary').innerText = "기본가: " + info.price;
        document.getElementById('color-data').innerText = info.colors;
        
        const pdfBtn = document.getElementById('price-pdf-btn');
        pdfBtn.onclick = () => window.open(info.pdf || 'https://www.hyundai.com/kr/ko/e/search?searchKeyword=' + carName, '_blank');
    }
