<script>
    const data = {
        genesis: {
            name: "제네시스 라인업",
            cars: ["G70", "G80", "G90", "GV70", "GV80"],
            url: "https://www.genesis.com/kr/ko/view-delivery/estimate/models.html"
        },
        hyundai: {
            name: "현대자동차 라인업",
            cars: ["아반떼", "쏘나타", "그랜저", "베뉴", "코나", "투싼", "싼타페", "팰리세이드", "포터", "스타리아"],
            url: "https://www.hyundai.com/kr/ko/e/vehicles/build-your-own"
        }
    };

    let selectedBrand = '';

    function showBrand(brand) {
        selectedBrand = brand;
        document.getElementById('main-screen').className = 'hidden';
        document.getElementById('list-screen').className = '';
        document.getElementById('brand-name').innerText = data[brand].name;
        
        const listContainer = document.getElementById('car-list');
        listContainer.innerHTML = '';
        
        data[brand].cars.forEach(car => {
            const div = document.createElement('div');
            div.className = 'car-item';
            div.innerText = car;
            // 각 차종별 상세 정보(가격/제원) 페이지로 이동
            div.onclick = () => {
                const searchUrl = `https://www.hyundai.com/kr/ko/e/search?searchKeyword=${encodeURIComponent(car)}`;
                window.open(searchUrl, '_blank');
            };
            listContainer.appendChild(div);
        });
    }

    function goHome() {
        document.getElementById('main-screen').className = '';
        document.getElementById('list-screen').className = 'hidden';
    }

    function openBuild() {
        window.open(data[selectedBrand].url, '_blank');
    }

    function shareCard() {
        const myInfo = "안녕하세요! 현대자동차 카마스터입니다.\n상담이 필요하시면 언제든 연락주세요!";
        if (navigator.share) {
            navigator.share({ title: '현대차 상담 명함', text: myInfo });
        } else {
            alert("명함 정보: " + myInfo);
        }
    }
</script>
